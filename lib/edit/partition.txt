#   N:<partition>

#
# Drop profile lines:
#   PF:w:a:j:s:potion:herb:gem:staff:misc:tunneling:allow_damaged
#     Floor / generated-on-ground profile for this partition.
#   PC:w:a:j:s:potion:herb:gem:staff:misc:tunneling:allow_damaged
#     Chest contents profile for this partition.
#   PM:w:a:j:s:potion:herb:gem:staff:misc:tunneling:allow_damaged
#     Monster-drop profile for this partition.
#     allow_damaged:
#       1 = allow damaged items to participate in this profile.
#       0 = keep damaged items out unless explicitly requested elsewhere.
#
# Placement / count lines:
#   FL:allow_floor_drops
#     allow_floor_drops:
#       1 = allow normal floor/object scatter in this partition.
#       0 = disable generated floor drops in this partition.
#
#   MB:base_monster_scale_num:base_monster_scale_den
#     Base room-driven monster count multiplier.
#     Base count is roughly (room_count + 1d(room_count)) / 2,
#     then multiplied by num / den.
#
#   MF:direct_floor_monster_divisor:min:max
#     Extra monsters added directly from partition floor area.
#     Formula:
#       floor_count / divisor, clamped to min..max.
#     Use divisor 0 to disable this bonus.
#
#   MD:depth_bonus_divisor:min:max:scale_pct_at_depth_20:hard_cap_divisor
#     Extra monsters from floor area that also scale with depth.
#     Formula:
#       floor_count / depth_bonus_divisor, clamped to min..max,
#       then scaled up with depth until depth 20,
#       then capped by floor_count / hard_cap_divisor.
#     scale_pct_at_depth_20:
#       Final percentage reached by depth 20.
#       100 = no extra depth scaling.
#     Use depth_bonus_divisor 0 to disable this bonus.
#
#   OD:room_object_divisor:corridor_object_divisor
#     Generated object counts derived from total monsters in the partition.
#     Formula:
#       room_objects = total_monsters / room_object_divisor
#       corridor_objects = total_monsters / corridor_object_divisor
#     Use divisor 0 to disable one side.
#
#   MT:floor_divisor:min:max:min_depth
#     Metal pieces generated from partition floor area.
#     Formula:
#       floor_count / floor_divisor, clamped to min..max.
#     Use divisor 0 to disable metal drops in this partition.
#     min_depth:
#       Dungeon depth where metal drops start.
#
# Narrative lines:
#   T:text
#     Discovery banner/message shown the first time this partition type is
#     encountered in a run.
#   BI:text
#   BF:text
#   BP:text
#     Big-cave discovery text for ice, fire, and poison variants.
#
